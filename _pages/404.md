<script>
  // Old academicpages URLs (e.g. /publication/2026-04-26-tft) are sent to the matching overview page.
  (function () {
    var rules = [
      ["/publication/", "/publications/"],
      ["/workinprogress/", "/workinprogress/"],
      ["/presentations/", "/presentations/"],
      ["/teaching/", "/teaching/"],
      ["/portfolio/", "/portfolio/"],
    ];
    var path = window.location.pathname;
    for (var i = 0; i < rules.length; i++) {
      if (path.indexOf(rules[i][0]) === 0 && path.length > rules[i][0].length) {
        window.location.replace(rules[i][1]);
        return;
      }
    }
  })();
</script>
